import React, { useState } from "react";
import {
  View,
  Text,
  TextInput,
  Button,
  Image,
  StyleSheet,
  TouchableOpacity,
  ActivityIndicator,
  ScrollView,
  ImageBackground,
  Alert,
} from "react-native";
import * as ImagePicker from "expo-image-picker";
import axios from "axios";

// 🔐 ONLY VALID LICENSE
const LICENSE_KEY = "KAYY-888-666";

// 🔑 PUT YOUR API KEY HERE
const OPENAI_API_KEY = "YOUR_OPENAI_API_KEY";

export default function App() {
  const [isUnlocked, setIsUnlocked] = useState(false);
  const [inputKey, setInputKey] = useState("");
  const [image, setImage] = useState(null);
  const [loading, setLoading] = useState(false);
  const [result, setResult] = useState(null);

  // 🔐 LOGIN
  const handleLogin = () => {
    if (inputKey.trim() === LICENSE_KEY) {
      setIsUnlocked(true);
    } else {
      Alert.alert("❌ Invalid License Key");
    }
  };

  // 📸 UPLOAD IMAGE
  const pickImage = async () => {
    const permission =
      await ImagePicker.requestMediaLibraryPermissionsAsync();

    if (!permission.granted) {
      Alert.alert("Permission required");
      return;
    }

    let result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      quality: 1,
    });

    if (!result.canceled) {
      setImage(result.assets[0].uri);
      setResult(null);
    }
  };

  // 📷 TAKE PHOTO
  const takePhoto = async () => {
    const permission = await ImagePicker.requestCameraPermissionsAsync();

    if (!permission.granted) {
      Alert.alert("Camera permission required");
      return;
    }

    let result = await ImagePicker.launchCameraAsync({
      quality: 1,
    });

    if (!result.canceled) {
      setImage(result.assets[0].uri);
      setResult(null);
    }
  };

  // 🤖 AI ANALYSIS (LOCKED)
  const analyzeChart = async () => {
    // 🚫 HARD LOCK
    if (!isUnlocked) {
      Alert.alert("🔒 Access Denied", "Enter valid license first");
      return;
    }

    if (!image) {
      Alert.alert("Upload a chart first");
      return;
    }

    setLoading(true);

    try {
      const response = await axios.post(
        "https://api.openai.com/v1/responses",
        {
          model: "gpt-4.1-mini",
          input: [
            {
              role: "user",
              content: [
                {
                  type: "input_text",
                  text: `
You are TRADECORE EA V1.0, a professional forex trading AI.

Analyze this chart and respond ONLY in this format:

Signal: BUY / SELL / NO ENTRY
Entry: ...
Take Profit: ...
Stop Loss: ...
Timeframe: ...
Strength: ...
Analysis: ...
                  `,
                },
                {
                  type: "input_image",
                  image_url: image,
                },
              ],
            },
          ],
        },
        {
          headers: {
            Authorization: `Bearer ${OPENAI_API_KEY}`,
            "Content-Type": "application/json",
          },
        }
      );

      const output =
        response.data.output[0].content[0].text || "No result";

      setResult(output);
    } catch (error) {
      console.log(error);
      Alert.alert("❌ Analysis failed");
    }

    setLoading(false);
  };

  // 🔐 LICENSE SCREEN (BLOCKS EVERYTHING)
  if (!isUnlocked) {
    return (
      <ImageBackground
        source={require("./assets/wallpaper.jpg")}
        style={styles.bg}
        blurRadius={4}
      >
        <View style={styles.overlay}>
          <Text style={styles.title}>TRADECORE EA V1.0</Text>

          <TextInput
            style={styles.input}
            placeholder="Enter License Key"
            placeholderTextColor="#ccc"
            value={inputKey}
            onChangeText={setInputKey}
          />

          <TouchableOpacity style={styles.unlockBtn} onPress={handleLogin}>
            <Text style={styles.btnText}>UNLOCK</Text>
          </TouchableOpacity>
        </View>
      </ImageBackground>
    );
  }

  // 📊 MAIN APP
  return (
    <ImageBackground
      source={require("./assets/wallpaper.jpg")}
      style={styles.bg}
      blurRadius={2}
    >
      <ScrollView contentContainerStyle={styles.overlay}>
        <Text style={styles.title}>TRADECORE EA V1.0</Text>

        <TouchableOpacity style={styles.btn} onPress={pickImage}>
          <Text style={styles.btnText}>Upload Chart</Text>
        </TouchableOpacity>

        <TouchableOpacity style={styles.btn} onPress={takePhoto}>
          <Text style={styles.btnText}>Take Photo</Text>
        </TouchableOpacity>

        {image && <Image source={{ uri: image }} style={styles.image} />}

        <TouchableOpacity style={styles.analyzeBtn} onPress={analyzeChart}>
          <Text style={styles.btnText}>ANALYZE CHART</Text>
        </TouchableOpacity>

        {loading && <ActivityIndicator size="large" color="#fff" />}

        {result && (
          <View style={styles.resultBox}>
            <Text style={{ color: "#fff" }}>{result}</Text>
          </View>
        )}
      </ScrollView>
    </ImageBackground>
  );
}

    flex: 1,
  },
  overlay: {
    flexGrow: 1,
    justifyContent: "center",

  },

  },

  },

    borderRadius: 5,
  },
  unlockBtn: {

    padding: 15,
    width: "100%",
    alignItems: "center",
    borderRadius: 5,
  },
  analyzeBtn: {
    backgroundColor: "#00c853",

  },
});
